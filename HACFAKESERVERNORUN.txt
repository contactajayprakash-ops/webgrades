// ====================
// Helper: Fetch with retry
// ====================
const fetchWithRetry = async (fetch, url, options = {}, retries = 3) => {
  for (let i = 0; i < retries; i++) {
    try {
      const res = await fetch(url, options);
      return res;
    } catch (err) {
      if (i === retries - 1) throw err;
      await new Promise((resolve) => setTimeout(resolve, 1000));
    }
  }
};

// ====================
// Shared Login Function
// Returns: { fetch, userName }
// ====================
const loginHAC = async (username, password) => {
  const cookieJar = new CookieJar();
  const fetch = fetchCookie(fetchOrig, cookieJar);

  const loginUrl =
    "https://hac.friscoisd.org/HomeAccess/Account/LogOn?ReturnUrl=%2fHomeAccess%2f";
  const loginPage = await fetchWithRetry(fetch, loginUrl, {
    headers: { "User-Agent": "Mozilla/5.0" },
    redirect: "manual",
  });
  const loginHtml = await loginPage.text();
  const loginDom = new JSDOM(loginHtml);

  const tokenInput = loginDom.window.document.querySelector(
    'input[name="__RequestVerificationToken"]',
  );
  if (!tokenInput) throw new Error("Login token not found");

  const token = tokenInput.value;
  const formData = new URLSearchParams();
  formData.append("__RequestVerificationToken", token);
  formData.append("LogOnDetails.UserName", username);
  formData.append("LogOnDetails.Password", password);
  formData.append("Database", "10");
  formData.append("VerificationOption", "UsernamePassword");

  const loginRes = await fetch(loginUrl, {
    method: "POST",
    body: formData,
    redirect: "manual",
    headers: {
      "Content-Type": "application/x-www-form-urlencoded",
      "User-Agent": "Mozilla/5.0",
      Referer: loginUrl,
    },
  });

  if (loginRes.status !== 302) throw new Error("Login failed");

  const homeUrl = new URL(loginRes.headers.get("location"), loginUrl).href;
  const homePage = await fetch(homeUrl, {
    headers: { "User-Agent": "Mozilla/5.0", Referer: loginUrl },
  });
  const homeHtml = await homePage.text();
  const homeDom = new JSDOM(homeHtml);

  const userNameElement = homeDom.window.document.querySelector(
    ".sg-menu-element-identity span",
  );

  const userName = userNameElement?.textContent.trim() || "Unknow n User";

  return { fetch, userName };
};

// ====================
// Warm-session cache
// Keep each user's logged-in cookie jar in memory briefly so repeat refreshes
// (the "everyone checks the new grade at once" rush) skip the 3-trip login.
// ====================
const sessions = new Map(); // username -> { fetch, userName, ts }
const SESSION_TTL_MS = 3 * 60 * 1000; // 3 minutes

const getSession = async (username, password, fresh = false) => {
  const cached = sessions.get(username);
  // Reuse only if the password matches the one that created the session, so a
  // warm session can never be served to a request with the wrong password.
  if (
    !fresh &&
    cached &&
    cached.password === password &&
    Date.now() - cached.ts < SESSION_TTL_MS
  ) {
    return cached;
  }
  const s = await loginHAC(username, password);
  const entry = {
    fetch: s.fetch,
    userName: s.userName,
    password,
    ts: Date.now(),
  };
  sessions.set(username, entry);
  return entry;
};

// A page that still shows the login token means the session expired and HAC
// bounced us back to the sign-in screen — signal so the caller can re-login.
class SessionExpired extends Error {
  constructor() {
    super("SESSION_EXPIRED");
    this.name = "SessionExpired";
  }
}
const assertLoggedIn = (doc) => {
  // The sign-in page is the only page with the username field — its presence
  // means our session expired and HAC bounced us back to LogOn.
  if (doc.querySelector('input[name="LogOnDetails.UserName"]')) {
    throw new SessionExpired();
  }
};

// ====================
// Helper: Parse Quarter Dropdown
// ====================
const getQuarterValue = (iframeDom, quarter) => {
  const options = Array.from(
    iframeDom.window.document.querySelectorAll(
      "#plnMain_ddlReportCardRuns option",
    ),
  );

  // Strip "Q" if user sent "Q1"
  const qNum = quarter.replace(/\D/g, "");

  // Find option where text matches or starts with the number (matches "1" or "1-2026")
  const selected = options.find(
    (o) =>
      o.textContent.trim() === qNum || o.textContent.trim().startsWith(qNum),
  );

  return selected ? selected.value : options[1]?.value;
};

// ====================
// Fetch Classwork with Quarter Support
// Same flow as the original (fetch landing -> iframe -> optional quarter POST);
// the speed win comes from reusing the logged-in session, not from skipping the
// landing fetch (which may set session state the quarter POST relies on).
// ====================
const getClassworkDom = async (fetch, quarter = null) => {
  const classworkUrl = "https://hac.friscoisd.org/HomeAccess/Classes/Classwork";
  const classworkPage = await fetch(classworkUrl, {
    headers: { "User-Agent": "Mozilla/5.0" },
  });
  const classworkHtml = await classworkPage.text();
  const classworkDom = new JSDOM(classworkHtml);
  assertLoggedIn(classworkDom.window.document);

  const iframe = classworkDom.window.document.querySelector("iframe");
  if (!iframe) throw new Error("Classwork iframe not found");

  const iframeUrl = iframe.src.startsWith("http")
    ? iframe.src
    : "https://hac.friscoisd.org" + iframe.src;

  const iframePage = await fetch(iframeUrl, {
    headers: { "User-Agent": "Mozilla/5.0", Referer: classworkUrl },
  });

  let iframeHtml = await iframePage.text();
  let iframeDom = new JSDOM(iframeHtml);
  assertLoggedIn(iframeDom.window.document);

  if (quarter) {
    const quarterValue = getQuarterValue(iframeDom, quarter);
    const doc = iframeDom.window.document;

    const formData = new URLSearchParams();

    // Mimic the __doPostBack('ctl00$plnMain$btnRefreshView','') call seen in the HTML
    formData.append("__EVENTTARGET", "ctl00$plnMain$btnRefreshView");
    formData.append("__EVENTARGUMENT", "");
    formData.append(
      "__VIEWSTATE",
      doc.querySelector("#__VIEWSTATE")?.value || "",
    );
    formData.append(
      "__VIEWSTATEGENERATOR",
      doc.querySelector("#__VIEWSTATEGENERATOR")?.value || "",
    );
    formData.append(
      "__EVENTVALIDATION",
      doc.querySelector("#__EVENTVALIDATION")?.value || "",
    );

    // Required fields for the Frisco ISD view
    formData.append("ctl00$plnMain$ddlReportCardRuns", quarterValue);
    formData.append("ctl00$plnMain$ddlClasses", "ALL");
    formData.append("ctl00$plnMain$ddlCompetencies", "ALL");
    formData.append("ctl00$plnMain$ddlOrderBy", "Class");

    const postRes = await fetch(iframeUrl, {
      method: "POST",
      headers: {
        "Content-Type": "application/x-www-form-urlencoded",
        "User-Agent": "Mozilla/5.0",
        Referer: iframeUrl,
      },
      body: formData.toString(),
    });

    iframeHtml = await postRes.text();
    iframeDom = new JSDOM(iframeHtml);
  }

  return iframeDom;
};

// HAC renders a per-class category-summary table (Category | Weight | Points |
// Percent) using the SAME `tr.sg-asp-table-data-row` class as real assignments,
// so the parser scoops those rows up too — they show up as fake assignments with
// a bare-number "name" ("100.00", "200.00") and a percentage "category"
// ("88.000%"). They only appear in the active quarter's view. Drop them at the
// source so no consumer (snapshot, /data, /batch, the "Recently posted" feed)
// ever sees them. Bare-number name is the discriminator (same heuristic the push
// poller already trusts). Real assignments have text names.
const isSubtotalName = (n) => /^\s*-?\d+(\.\d+)?\s*$/.test(String(n || ""));

// Parse the classwork DOM into { assignmentsData }
const parseClasswork = (classDom) => {
  const doc = classDom.window.document;
  const assignmentClasses = doc.querySelectorAll(".AssignmentClass");

  const assignmentsData = Array.from(assignmentClasses).map((classDiv) => {
    const courseName =
      classDiv
        .querySelector(".sg-header-heading")
        ?.textContent.replace(/\s+/g, " ")
        .trim() || "Unknown";

    const rawGradeText =
      classDiv
        .querySelector(".sg-header-heading.sg-right")
        ?.textContent.trim() || "";
    const gradeMatch = rawGradeText.match(/(\d+(\.\d+)?%)/);
    const overallAverage = gradeMatch ? gradeMatch[0] : "N/A";

    const assignmentRows = classDiv.querySelectorAll(
      "tr.sg-asp-table-data-row",
    );
    const assignments = Array.from(assignmentRows).map((row) => {
      const cells = row.querySelectorAll("td");
      return {
        assignmentName:
          cells[2]?.querySelector("a")?.textContent.trim() ||
          cells[2]?.textContent.trim(),
        category: cells[3]?.textContent.trim(),
        dateDue: cells[0]?.textContent.trim(),
        grade: cells[4]?.textContent.trim(),
        totalPoints: cells[5]?.textContent.trim(),
      };
    }).filter((a) => !isSubtotalName(a.assignmentName)); // drop category-summary rows

    return { courseName, overallAverage, assignments };
  });

  return { assignmentsData };
};

const getTranscriptDoc = async (fetch, cache = {}) => {

const parseRank = (doc) => {
  const gpa =
    doc
      .querySelector("#plnMain_rpTranscriptGroup_lblGPACum1")
      ?.textContent.trim() || null;
  const rankRaw =
    doc
      .querySelector("#plnMain_rpTranscriptGroup_lblGPARank1")
      ?.textContent.trim() || null;
  const rankParts = rankRaw ? rankRaw.split(" / ") : [null, null];
  return { gpa, rank: rankParts[0] || null, outOf: rankParts[1] || null };
};

const parseTranscript = (doc) => {
  const groups = Array.from(doc.querySelectorAll(".sg-transcript-group"));
  const transcript = groups.map((group) => {
    const year =
      group.querySelector('[id*="lblYearValue"]')?.textContent.trim() || null;
    const grade =
      group.querySelector('[id*="lblGradeValue"]')?.textContent.trim() || null;
    const building =
      group.querySelector('[id*="lblBuildingValue"]')?.textContent.trim() ||
      null;
    const rows = Array.from(group.querySelectorAll("tr.sg-asp-table-data-row"));
    const courses = rows.map((row) => {
      const cells = row.querySelectorAll("td");
      return {
        courseCode: cells[0]?.textContent.trim() || null,
        description: cells[1]?.textContent.trim() || null,
        sem1: cells[2]?.textContent.trim() || null,
        sem2: cells[3]?.textContent.trim() || null,
        final: cells[4]?.textContent.trim() || null,
        credit: cells[5]?.textContent.trim() || null,
      };
    });
    const totalCredit =
      group.querySelector('[id*="LblTCreditValue"]')?.textContent.trim() ||
      null;
    return { year, grade, building, totalCredit, courses };
  });
  return { transcript };
};

const scrapeAttendance = async (fetch) => {
  const attendanceUrl =
    "https://hac.friscoisd.org/HomeAccess/Content/Attendance/MonthlyView.aspx";
  const attendancePage = await fetch(attendanceUrl, {
    headers: { "User-Agent": "Mozilla/5.0" },
  });
  const document = new JSDOM(await attendancePage.text()).window.document;
  assertLoggedIn(document);
  const calendarCells = document.querySelectorAll("#plnMain_cldAttendance td");
  const days = Array.from(calendarCells)
    .map((cell) => {
      const day = Number(cell.textContent.trim());
      if (isNaN(day)) return null;
      const style = cell.getAttribute("style") || "";
      const colorMatch = style.match(/background-color:\s*(#[A-Fa-f0-9]+)/);
      return {
        day,
        color: colorMatch?.[1] || null,
        tooltip: cell.getAttribute("title") || null,
      };
    })
    .filter(Boolean);
  return {
    month: document
      .querySelector(".sg-asp-calendar-header td:nth-child(2)")
      ?.textContent.trim(),
    days,
  };
};

const scrapeWeek = async (fetch) => {
  const weekUrl = "https://hac.friscoisd.org/HomeAccess/Home/WeekView";
  const weekPage = await fetchWithRetry(fetch, weekUrl);
  const dom = new JSDOM(await weekPage.text());
  const rows = dom.window.document.querySelectorAll(
    ".sg-homeview-table tbody tr",
  );
  const classes = Array.from(rows).map((row) => {
    const course = row.querySelector("#courseName")?.textContent.trim();
    const teacher = row.querySelector("#staffName")?.textContent.trim();
    const periodMatch = row.textContent.match(/Per:\s*(\w+)/);
    const period = periodMatch ? periodMatch[1] : null;
    const assignments = Array.from(row.querySelectorAll("td")).flatMap(
      (cell, index) =>
        Array.from(cell.querySelectorAll("#courseAssignmentDescription")).map(
          (a) => ({
            title: a.textContent.trim(),
            grade: cell.querySelector(".sg-right")?.textContent.trim() || null,
            dayIndex: index,
          }),
        ),
    );
    return { course, teacher, period, assignments };
  });
  return { classes };
};

// ====================
// One dispatcher used by BOTH /data and /batch. `cache` is per-request so a
// batch can share the classwork iframe URL + transcript page across items.
// ====================
const scrapeType = async (fetch, { type, quarter, date } = {}, cache = {}) => {
  switch (type) {
    case "class":
      return parseClasswork(await getClassworkDom(fetch, quarter));
    case "rank":
      return parseRank(await getTranscriptDoc(fetch, cache));
    case "transcript":
      return parseTranscript(await getTranscriptDoc(fetch, cache));
    case "schedule":
      return scrapeSchedule(fetch);
    case "attendance":
      return scrapeAttendance(fetch);
    case "week":
      return scrapeWeek(fetch);
    case "ipr":
      return scrapeIpr(fetch, date);
    default:
      throw new Error("Invalid type requested");
  }
};

// Run one scrape, transparently re-logging-in once if the warm session expired.
const scrapeWithSession = async (username, password, request, cache) => {
  let session = await getSession(username, password);
  try {
    return {
      data: await scrapeType(session.fetch, request, cache),
      userName: session.userName,
    };
  } catch (err) {
    if (err instanceof SessionExpired) {
      // session went stale — drop the per-request page cache, re-login, retry once
      delete cache.transcriptDoc;
      session = await getSession(username, password, true);
      return {
        data: await scrapeType(session.fetch, request, cache),
        userName: session.userName,
      };
    }
    throw err;
  }
};

// Batch endpoint — log in ONCE, scrape every requested resource in that single
// session, return them all together. This is what makes a grade-drop refresh
// fast: one round-trip instead of ~9 serial logins.
//   body: { username, password, requests: [{ type, quarter?, date? }, ...] }
//   resp: { success, userName, results: [{ type, quarter?, success, data?, message? }] }
app.post("/batch", async (req, res) => {
  const { username, password, requests } = req.body;
  if (!Array.isArray(requests) || requests.length === 0) {
    return res.json({ success: false, message: "No requests provided." });
  }
  try {
    // Prime the session once up front (also surfaces bad credentials early).
    const primed = await getSession(username, password);
    registerSnapshot(username, password); // seen just now → keep its snapshot warm
    let userName = primed.userName;
    const cache = {}; // shared across this batch: iframe URL + transcript page

    const results = [];
    for (const r of requests) {
      try {
        const out = await scrapeWithSession(username, password, r, cache);
        userName = out.userName || userName;
        results.push({ ...r, success: true, data: out.data });
      } catch (err) {
        results.push({ ...r, success: false, message: err.message });
      }
    }

    const okCount = results.filter((x) => x.success).length;
    console.log(
      `[${ctTime()}] BATCH — username=${username} ${okCount}/${requests.length} ok`,
    );
    res.json({ success: true, userName, results });
  } catch (err) {
    console.log(
      `[${ctTime()}] BATCH FAILED — username=${username} reason=${err.message}`,
    );
    res.json({ success: false, message: err.message });
  }
});

// Keys come from the environment so no secret lives in source:

const VAPID_PRIVATE = process.env.VAPID_PRIVATE_KEY || "";
const VAPID_SUBJECT = process.env.VAPID_SUBJECT || "mailto:webgrades@friscoisd.invalid";
const PUSH_ENABLED = !!(VAPID_PUBLIC && VAPID_PRIVATE);
if (PUSH_ENABLED) {
  webpush.setVapidDetails(VAPID_SUBJECT, VAPID_PUBLIC, VAPID_PRIVATE);
  console.log(`[${ctTime()}] Web Push ENABLED`);
} else {
  console.log(`[${ctTime()}] Web Push DISABLED — set VAPID_PUBLIC_KEY / VAPID_PRIVATE_KEY`);

// Persisted subscriber store: username -> { password, kinds, subs:[subscription],
// lastSeen:{ "course|assignment": grade } }. Plaintext by choice — lock the file
// to the owner (the writes below use mode 0600).
const PUSH_FILE = process.env.PUSH_STORE || "./push-subs.json";
let pushStore = {};
try {
  if (fs.existsSync(PUSH_FILE)) pushStore = JSON.parse(fs.readFileSync(PUSH_FILE, "utf8")) || {};
} catch (e) { console.log(`[${ctTime()}] push store load failed: ${e.message}`); }
const savePushStore = () => {
  try { fs.writeFileSync(PUSH_FILE, JSON.stringify(pushStore), { mode: 0o600 }); }
  catch (e) { console.log(`[${ctTime()}] push store save failed: ${e.message}`); }
};

const isAssessment = (c) => /assess|aol/i.test(c || "");
const isProgress = (c) => /progress|prog|\bpc\b/i.test(c || "");
const kindMatches = (category, kinds) => {
  const aol = isAssessment(category), pc = isProgress(category);
  if (kinds === "both") return aol || pc;
  if (kinds === "pc") return pc;
  return aol; // default: Assessments (AOL) only
};
const gradedVal = (g) => g != null && String(g).trim() !== "" && /\d/.test(String(g));
// Normalize a grade for STABLE comparison: HAC returns inconsistent decimals
// ("97" / "97.0" / "97.00"), which made the string compare read unchanged grades
// as new and re-notify every poll. Compare/store the numeric form ("97").
const normGrade = (g) => { const n = parseFloat(String(g)); return Number.isFinite(n) ? String(n) : String(g).trim(); };
// (isSubtotalName is defined near parseClasswork — these junk rows are now
// filtered at the parser, so the poller rarely sees one, but the guard stays.)
// The dedup key now includes CATEGORY. HAC lists the same assignment name under
// both a Progress and an Assessment category (e.g. "1D Motion Math" = 70 PC AND
// 97 AOL); the old `course|name` key collided, so the two values overwrote each
// other and re-notified every other poll. Reset each baseline ONCE (keyVer) so
// the first poll re-captures under the new keys silently.
for (const u of Object.keys(pushStore)) {
  const e = pushStore[u];
  if (e && e.keyVer !== 2) { e.lastSeen = {}; e.keyVer = 2; }
}
savePushStore();

// Register/refresh a device. Verifies the credentials before storing them.
app.post("/push/subscribe", async (req, res) => {
  const { username, password, kinds, subscription } = req.body || {};
  if (!username || !subscription || !subscription.endpoint) {
    return res.json({ success: false, message: "missing fields" });
  }
  try {
    if (password) await getSession(username, password); // reject bad creds up front
    const entry = pushStore[username] || { password, kinds: "aol", subs: [], lastSeen: {} };
    if (password) entry.password = password;
    if (kinds) entry.kinds = kinds;
    entry.subs = (entry.subs || []).filter((s) => s.endpoint !== subscription.endpoint);
    entry.subs.push(subscription);
    pushStore[username] = entry;
    savePushStore();
    console.log(`[${ctTime()}] PUSH SUBSCRIBE — username=${username} kinds=${entry.kinds} devices=${entry.subs.length}`);
    res.json({ success: true });
  } catch (err) {
    res.json({ success: false, message: err.message });
  }
});

app.post("/push/unsubscribe", (req, res) => {
  const { username, endpoint } = req.body || {};
  const entry = pushStore[username];
  if (entry) {
    entry.subs = (entry.subs || []).filter((s) => s.endpoint !== endpoint);
    if (!entry.subs.length) delete pushStore[username];
    savePushStore();
  }
  res.json({ success: true });
});

// Send a test notification to a user's devices (verify end-to-end). Gated on
// the correct password so it can't be used to spam someone.
app.post("/push/test", async (req, res) => {
  const { username, password } = req.body || {};
  const entry = pushStore[username];
  if (!entry || !(entry.subs || []).length) {
    return res.json({ success: false, message: "no subscription for this user" });
  }
  try {
    if (!password || password !== entry.password) {
      return res.json({ success: false, message: "bad password" });
    }
    await sendPush(entry, { title: "WebGrades test", body: "Push notifications are working ✅", url: "/" });
    savePushStore();
    res.json({ success: true, devices: entry.subs.length });
  } catch (err) {
    res.json({ success: false, message: err.message });
  }
});

// Trigger one poll cycle on demand (gated by password) — to verify dedup
// end-to-end without waiting for the 15-minute timer.
app.post("/push/poll", async (req, res) => {
  const { username, password } = req.body || {};
  const entry = pushStore[username];
  if (!entry) return res.json({ success: false, message: "no subscription" });
  if (!password || password !== entry.password) {
    return res.json({ success: false, message: "bad password" });
  }
  await pollOnce();
  res.json({ success: true });
});

// Deliver a payload to all of a user's devices, pruning dead subscriptions.
const sendPush = async (entry, payload) => {
  const body = JSON.stringify(payload);
  const alive = [];
  for (const sub of entry.subs || []) {
    try { await webpush.sendNotification(sub, body); alive.push(sub); }
    catch (err) {
      if (err.statusCode === 404 || err.statusCode === 410) continue; // gone — drop it
      alive.push(sub); // transient — keep, retry next cycle
    }
  }
  entry.subs = alive;
};

// Poll each subscriber's CURRENT grades, diff against lastSeen, push the new ones.
const POLL_INTERVAL_MS = Number(process.env.PUSH_POLL_MS || 15 * 60 * 1000); // 15 min
let polling = false;
const inActiveHours = () => {
  // Poll waking hours EVERY day — teachers post grades in the evening and on
  // weekends too, so a Mon–Fri 9-to-5 gate misses most real drops. Coarse
  // Central-Time gate (offset assumes CDT; close enough for a window this wide).
  const ct = new Date(Date.now() - 5 * 60 * 60 * 1000);
  const hr = ct.getUTCHours();
  return hr >= 6 && hr < 23; // 6am–11pm CT, all 7 days
};

const pollOnce = async () => {
  if (!PUSH_ENABLED || polling) return;
  const users = Object.keys(pushStore);
  if (!users.length) return;
  polling = true;
  try {
    for (const username of users) {
      const entry = pushStore[username];
      if (!entry || !entry.password || !(entry.subs || []).length) continue;
      try {
        const { data } = await scrapeWithSession(username, entry.password, { type: "class" }, {});
        const firstSeen = !entry.lastSeen || !Object.keys(entry.lastSeen).length;
        const nextSeen = { ...(entry.lastSeen || {}) };
        const events = [];
        for (const c of data.assignmentsData || []) {
          for (const a of c.assignments || []) {
            if (!gradedVal(a.grade)) continue;
            if (isSubtotalName(a.assignmentName)) continue; // skip HAC subtotal junk rows
            const key = `${c.courseName}|${a.category}|${a.assignmentName}`;
            const nv = normGrade(a.grade); // normalized: stable across HAC decimal drift
            if (entry.lastSeen && entry.lastSeen[key] === nv) continue;
            nextSeen[key] = nv;
            if (kindMatches(a.category, entry.kinds || "aol")) {
              events.push({ course: c.courseName, name: a.assignmentName || "Assignment", grade: String(a.grade).trim() });
            }
          }
        }
        entry.lastSeen = nextSeen;
        // Never blast the whole gradebook the first time we poll this user.
        if (!firstSeen && events.length) {
          const line = (e) => `${e.course} — ${e.name}: ${e.grade}`;
          const payload = events.length === 1
            ? { title: "New grade posted", body: line(events[0]), url: "/" }
            : { title: `${events.length} new grades posted`, body: events.slice(0, 4).map(line).join("\n"), url: "/" };
          await sendPush(entry, payload);
          console.log(`[${ctTime()}] PUSH SENT — username=${username} n=${events.length}`);
        }
      } catch (err) {
        console.log(`[${ctTime()}] PUSH POLL FAIL — username=${username} reason=${err.message}`);
      }
      await new Promise((r) => setTimeout(r, 1500)); // stagger — easy on HAC + the Pi
    }
    savePushStore();
  } finally {
    polling = false;
  }
};

if (PUSH_ENABLED) {
  setInterval(() => { if (inActiveHours()) pollOnce(); }, POLL_INTERVAL_MS);
  // One-time baseline shortly after boot so a reset/key-format change re-captures
  // silently (firstSeen) instead of waiting up to a full interval.
  setTimeout(() => pollOnce(), 8000);
  console.log(`[${ctTime()}] Web Push poller: every ${Math.round(POLL_INTERVAL_MS / 60000)}m, 6am–11pm CT`);
}

// ============================================================
// Server-side grade snapshots — the Pi scrapes each known student on a timer
// and writes a compressed snapshot to Firestore, so the browser can paint a
// full gradebook from ONE doc read on a cold open (empty cache / new device /
// borrowed Chromebook) instead of waiting on ~9 serial HAC scrapes.
//
// Independent of Web Push: a user can have fast loads without notifications and
// vice versa. A complete no-op unless a Firebase service account is provided
// (FIREBASE_SA -> a JSON key path), exactly the way push is a no-op without
// VAPID keys — the app still works with this feature entirely off.
// ============================================================
const SA_PATH = process.env.FIREBASE_SA || process.env.GOOGLE_APPLICATION_CREDENTIALS || "";
let firestore = null;
if (SA_PATH && fs.existsSync(SA_PATH)) {
  try {
    // Lazy import: only load firebase-admin when we actually have a service
    // account, so a Pi without the package (or with the feature off) is
    // unaffected. Use the modular subpath entry points (firebase-admin/app,
    // /firestore) — the clean ESM interface; the default `firebase-admin` import
    // doesn't expose `.credential` reliably under a dynamic import().
    const { initializeApp, cert } = await import("firebase-admin/app");
    const { getFirestore } = await import("firebase-admin/firestore");
    initializeApp({ credential: cert(JSON.parse(fs.readFileSync(SA_PATH, "utf8"))) });
    firestore = getFirestore();
    console.log(`[${ctTime()}] Snapshot sync ENABLED`);
  } catch (e) {
    console.log(`[${ctTime()}] Snapshot sync init FAILED — ${e.message}`);
  }
} else {
  console.log(`[${ctTime()}] Snapshot sync DISABLED — set FIREBASE_SA to a service account JSON`);
}
const SNAPSHOT_ENABLED = !!firestore;

// The doc id must match the browser's byte-for-byte (src/lib/cloudSync.js):
// SHA-256 hex of `${username} ${password}` with a single space and ''-fallbacks.
// If these drift, every client reads a missing doc and silently falls back to
// the slow path — guarded by scripts/test-credkey-parity.mjs.
const credKey = (username, password) =>
  crypto.createHash("sha256").update(`${username || ""} ${password || ""}`).digest("hex");

// Snapshot registry — WHO gets snapshotted. Independent of pushStore. Populated
// automatically on every verified /login and /batch (product decision: fast
// loading on by default for everyone). Plaintext creds, file locked to owner
// (0600) — the same trade-off already accepted for push-subs.json.
const SNAP_FILE = process.env.SNAPSHOT_STORE || "./snapshot-subs.json";
let snapStore = {};
try {
  if (fs.existsSync(SNAP_FILE)) snapStore = JSON.parse(fs.readFileSync(SNAP_FILE, "utf8")) || {};
} catch (e) { console.log(`[${ctTime()}] snapshot store load failed: ${e.message}`); }
const saveSnapStore = () => {
  try { fs.writeFileSync(SNAP_FILE, JSON.stringify(snapStore), { mode: 0o600 }); }
  catch (e) { console.log(`[${ctTime()}] snapshot store save failed: ${e.message}`); }
};

// Record/refresh an account in the snapshot registry. No-op when the feature is
// off, so nothing new is stored when there's no service account. A brand-new
// sign-in (no snapshot yet) is snapshotted RIGHT AWAY — async, non-blocking — so
// their fast-load doc exists for the next cold open instead of waiting up to a
// full poll cycle. `snapshotNow` is defined below; the setTimeout defers the ref.
const registerSnapshot = (username, password) => {
  if (!SNAPSHOT_ENABLED || !username || !password) return;
  const e = snapStore[username] || {};
  const isNew = !e.snapHash; // never snapshotted → prime it immediately
  e.password = password;
  e.seenAt = Date.now(); // last activity — drives poll tiering (#5)
  snapStore[username] = e;
  saveSnapStore();
  if (isNew) setTimeout(() => snapshotNow(username, password, "full"), 0);
};

// Same calendar guess the client uses (src/lib/courses.js guessCurrentQuarter),
// so the generic `class:{}` view aliases to the same quarter on both sides.
const guessCurrentQuarter = (d = new Date()) => {
  const m = d.getMonth();
  if (m >= 7 && m <= 9) return "1";
  if (m === 10 || m === 11) return "2";
  if (m >= 0 && m <= 2) return "3";
  return "4";
};

// Cache-key shape MUST match the client's keyOf(type, extra) in AuthContext.jsx:
//   `${type}:${JSON.stringify(rest)}`  ->  "class:{}", "class:{\"quarter\":\"3\"}", ...
const snapKey = (type, extra = {}) => `${type}:${JSON.stringify(extra)}`;

// The FULL resource set (client cache-map shape): `class` for each quarter, plus
// rank/transcript (one shared page) and schedule/attendance. The HOT set is just
// the current quarter — the only thing that moves between grading periods.
const fullResources = () => [
  ["class", { quarter: "1" }], ["class", { quarter: "2" }],
  ["class", { quarter: "3" }], ["class", { quarter: "4" }],
  ["rank", {}], ["transcript", {}], ["schedule", {}], ["attendance", {}],
];
const hotResources = (curQ) => [["class", { quarter: curQ }]];

// Scrape a set of resources in ONE session (the per-request page cache shares the
// transcript page across rank+transcript, like /batch). A failed resource is
// skipped, not fatal. Aliases the current quarter into the generic `class:{}`
// view (same HAC page) so the client's Dashboard/Agenda fallback is populated.
const scrapeResources = async (username, password, resources, curQ) => {
  const cache = {};
  const data = {};
  for (const [type, extra] of resources) {
    try {
      const { data: d } = await scrapeWithSession(username, password, { type, ...extra }, cache);
      data[snapKey(type, extra)] = d;
    } catch (e) { /* skip this resource; keep the rest */ }
  }
  const curData = data[snapKey("class", { quarter: curQ })];
  if (curData !== undefined) data[snapKey("class", {})] = curData;
  return data;
};

// Last full uncompressed snapshot per user, kept in memory so a HOT poll can
// patch just the current quarter onto it without re-scraping everything. Lost on
// restart → the first poll after boot is forced FULL to rebuild it.
const snapDataCache = new Map(); // username -> data object

// Write one gzipped snapshot doc per user at grades/<credKey>, but ONLY when the
// uncompressed payload actually changed (hash compare) — keeps us far inside the
// Firestore free tier. mode 'hot' patches the current quarter onto the last full
// data; 'full' re-scrapes everything. Doc: { codec, data(base64), hash,
// updatedAt, v }; updatedAt is the snapshot's real age (the client shows it).
const writeSnapshot = async (username, password, mode = "full") => {
  if (!SNAPSHOT_ENABLED) return { changed: false, bytes: 0, mode };
  const curQ = guessCurrentQuarter();
  const prior = snapDataCache.get(username);
  // Can't patch onto nothing — a hot poll with no in-memory base becomes full.
  const effMode = mode === "hot" && prior ? "hot" : "full";
  const scraped = await scrapeResources(
    username, password,
    effMode === "hot" ? hotResources(curQ) : fullResources(),
    curQ,
  );
  if (!Object.keys(scraped).length) return { changed: false, bytes: 0, mode: effMode }; // total failure
  const data = effMode === "hot" ? { ...prior, ...scraped } : scraped;
  snapDataCache.set(username, data);
  const entry = snapStore[username] || {};
  entry.password = password;
  if (effMode === "full") entry.lastFull = Date.now();
  const json = JSON.stringify(data);
  const hash = crypto.createHash("sha256").update(json).digest("hex");
  const gz = zlib.gzipSync(json);
  const bytes = gz.length;
  if (entry.snapHash === hash) { snapStore[username] = entry; return { changed: false, bytes, mode: effMode }; }
  await firestore.collection("grades").doc(credKey(username, password)).set({
    codec: "gzip", data: gz.toString("base64"), hash, updatedAt: Date.now(), v: 1,
  });
  entry.snapHash = hash;
  snapStore[username] = entry;
  saveSnapStore();
  return { changed: true, bytes, mode: effMode };
};

// One snapshot for one user, guarded so the same account is never scraped twice
// at once (an immediate on-sign-in snapshot vs the poll cycle picking up the same
// user). Logs + stamps lastSnapAt. Returns the write result, or null if skipped.
const snapInFlight = new Set(); // usernames currently being snapshotted
const snapshotNow = async (username, password, mode = "full") => {
  if (!SNAPSHOT_ENABLED || snapInFlight.has(username)) return null;
  snapInFlight.add(username);
  try {
    const res = await writeSnapshot(username, password, mode);
    if (snapStore[username]) { snapStore[username].lastSnapAt = Date.now(); saveSnapStore(); }
    console.log(`[${ctTime()}] SNAPSHOT — username=${username} mode=${res.mode} bytes=${res.bytes} changed=${res.changed ? "y" : "n"}`);
    return res;
  } catch (err) {
    console.log(`[${ctTime()}] SNAPSHOT FAIL — username=${username} reason=${err.message}`);
    return null;
  } finally {
    snapInFlight.delete(username);
  }
};

// ---- Poller tiering + concurrency (#5) ----
// Budget the poll against HAC, not Firestore: at a few hundred students a serial
// full-scrape loop would make thousands of hac.friscoisd.org logins a day from
// one IP — that gets noticed and blocked long before any free tier runs out. So
// tier by ACTIVITY (how recently the student used the app) and by RESOURCE
// (current grades often, everything else ~daily), run a small worker pool, and
// stagger. Every knob is env-configurable so intervals can be dialed back fast.
const SNAP_POLL_MS = Number(process.env.SNAPSHOT_POLL_MS || 15 * 60 * 1000); // cycle tick
const SNAP_FULL_TTL_MS = Number(process.env.SNAPSHOT_FULL_TTL_MS || 24 * 60 * 60 * 1000); // full re-scrape ~daily
const SNAP_ACTIVE_MS = Number(process.env.SNAPSHOT_ACTIVE_MS || 48 * 60 * 60 * 1000); // "active" window
const SNAP_DORMANT_MS = Number(process.env.SNAPSHOT_DORMANT_MS || 14 * 24 * 60 * 60 * 1000); // beyond → daily only
const SNAP_HOT_MS = Number(process.env.SNAPSHOT_HOT_MS || 15 * 60 * 1000); // active cadence
const SNAP_HOURLY_MS = Number(process.env.SNAPSHOT_HOURLY_MS || 60 * 60 * 1000); // dormant cadence
const SNAP_CONCURRENCY = Math.max(1, Number(process.env.POLL_CONCURRENCY || 2)); // JSDOM is memory-hungry on a 1GB Pi 3
const SNAP_STAGGER_MS = Number(process.env.SNAPSHOT_STAGGER_MS || 1500); // per-worker gap between users

// How often THIS user should be polled, by how recently they used the app.
const userCadence = (seenAt) => {
  const age = Date.now() - (seenAt || 0);
  if (age < SNAP_ACTIVE_MS) return SNAP_HOT_MS;     // active → check current grades every cycle
  if (age < SNAP_DORMANT_MS) return SNAP_HOURLY_MS; // dormant → hourly
  return SNAP_FULL_TTL_MS;                           // very dormant → daily
};

let snapPolling = false;
const snapshotPollOnce = async () => {
  if (!SNAPSHOT_ENABLED || snapPolling) return;
  const now = Date.now();
  const due = Object.keys(snapStore).filter((u) => {
    const e = snapStore[u];
    return e && e.password && now - (e.lastSnapAt || 0) >= userCadence(e.seenAt);
  });
  if (!due.length) return;
  snapPolling = true;
  try {
    let idx = 0;
    const worker = async () => {
      while (idx < due.length) {
        const username = due[idx++];
        const entry = snapStore[username];
        if (!entry || !entry.password) continue;
        // FULL when there's no in-memory base (e.g. after a restart) or the daily
        // TTL elapsed; otherwise a cheap HOT patch of just the current quarter.
        const needFull = !snapDataCache.has(username) || now - (entry.lastFull || 0) >= SNAP_FULL_TTL_MS;
        await snapshotNow(username, entry.password, needFull ? "full" : "hot"); // logs + guards + stamps
        await new Promise((r) => setTimeout(r, SNAP_STAGGER_MS)); // stagger — easy on HAC + the Pi
      }
    };
    await Promise.all(Array.from({ length: Math.min(SNAP_CONCURRENCY, due.length) }, () => worker()));
    saveSnapStore();
  } finally {
    snapPolling = false;
  }
};

// On-demand FULL snapshot for one user (gated by the correct password) — verify
// the writer end-to-end without waiting for the timer.
app.post("/snapshot/run", async (req, res) => {
  const { username, password } = req.body || {};
  if (!SNAPSHOT_ENABLED) return res.json({ success: false, message: "snapshots disabled" });
  const entry = snapStore[username];
  const pw = password || entry?.password;
  if (!pw) return res.json({ success: false, message: "no password" });
  try {
    await getSession(username, pw); // reject bad creds
    const res2 = await snapshotNow(username, pw, "full"); // guarded + logs + stamps
    if (!res2) return res.json({ success: false, message: "already snapshotting this user" });
    res.json({ success: true, changed: res2.changed, bytes: res2.bytes, mode: res2.mode });
  } catch (err) {
    res.json({ success: false, message: err.message });
  }
});

if (SNAPSHOT_ENABLED) {
  setInterval(() => { if (inActiveHours()) snapshotPollOnce(); }, SNAP_POLL_MS);
  setTimeout(() => snapshotPollOnce(), 12000); // baseline shortly after boot
  console.log(`[${ctTime()}] Snapshot poller: cycle ${Math.round(SNAP_POLL_MS / 60000)}m, concurrency ${SNAP_CONCURRENCY}, full≈daily, 6am–11pm CT`);
}

app.listen(3000, () => console.log("HAC API running on port 3000"));
